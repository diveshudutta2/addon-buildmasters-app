        function loadTasks() {
            const saved1 = localStorage.getItem('task1') === 'true';
            const saved2 = localStorage.getItem('task2') === 'true';
            const saved3 = localStorage.getItem('task3') === 'true';
            if (document.getElementById('task1')) document.getElementById('task1').checked = saved1;
            if (document.getElementById('task2')) document.getElementById('task2').checked = saved2;
            if (document.getElementById('task3')) document.getElementById('task3').checked = saved3;
            calculateScore();
        }

        function updateProgress() {
            if (document.getElementById('task1')) localStorage.setItem('task1', document.getElementById('task1').checked);
            if (document.getElementById('task2')) localStorage.setItem('task2', document.getElementById('task2').checked);
            if (document.getElementById('task3')) localStorage.setItem('task3', document.getElementById('task3').checked);
            calculateScore();
        }

        function calculateScore() {
            let score = 85;
            let pending = 3;
            if (document.getElementById('task1') && document.getElementById('task1').checked) { score += 5; pending--; }
            if (document.getElementById('task2') && document.getElementById('task2').checked) { score += 5; pending--; }
            if (document.getElementById('task3') && document.getElementById('task3').checked) { score += 5; pending--; }
            if (score > 100) score = 100;
            const scoreDisplay = document.getElementById('score-display');
            const pendingText = document.getElementById('pending-tasks-text');
            if (scoreDisplay) scoreDisplay.innerText = score + '%';
            if (pendingText) pendingText.innerText = pending === 0 ? 'All Tasks Completed Today! 🎉' : pending + ' Tasks Pending';
        }
